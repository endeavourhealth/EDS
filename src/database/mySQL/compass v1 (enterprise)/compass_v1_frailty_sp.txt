DELIMITER $$
CREATE PROCEDURE `get_frailty_flag`(
	_person_id bigint
)
BEGIN

	drop TEMPORARY table IF EXISTS tmp.frailty_patients ;
	CREATE TEMPORARY TABLE tmp.frailty_patients AS
    SELECT id as `patient_id`
    FROM patient
    WHERE person_id = _person_id;
    
    CREATE INDEX ix ON tmp.frailty_patients (patient_id);

	SELECT
		IF ((ob.snomed_concept_id = '713634000' AND result_value > '0.36')
		OR (ob.snomed_concept_id = '713636003' and result_value > '0.36')
		OR (ob.snomed_concept_id = '445414007' and result_value >= '7')
		OR (ob.snomed_concept_id IN ('925861000000102', '2010301000006105', '2010341000006107')), '3_SEVERE',
		
		IF ((ob.snomed_concept_id = '713634000' and result_value > '0.24' and result_value <= '0.36')
		OR (ob.snomed_concept_id = '713636003' and result_value > '0.24' and result_value <= '0.36')
		OR (ob.snomed_concept_id = '445414007' and result_value >= '6' and result_value < '7')
		OR (ob.snomed_concept_id IN ('925831000000107', '1839051000006106', '910301000006104', '248279007', '908431000006104', '909011000006109', '2010291000006109', '2010331000006102')), '2_MODERATE',

		IF ((ob.snomed_concept_id = '713634000' and result_value > '0.13' and result_value <= '0.24')
		OR (ob.snomed_concept_id = '713636003' and result_value > '0.13' and result_value <= '0.24')
		OR (ob.snomed_concept_id = '445414007' and result_value >= 5 and result_value < 6)
		OR (ob.snomed_concept_id IN ('925791000000100', '2010281000006106', '2010321000006100')), '1_MILD',
			
			null -- if frailty index is below the mild limit
		))) as `result`
	FROM observation ob
	INNER JOIN tmp.frailty_patients p
		ON p.patient_id = ob.patient_id
	AND ob.snomed_concept_id IN (
		'713634000', -- fiFrailtyIndex
		'713636003', -- fiFrailtyIndex2
		'445414007', -- cshaFrailtyScale
		'925861000000102', -- severeFrailty
		'2010301000006105', -- qFrailtySevereFrailty
		'2010341000006107', -- qFrailtySevereFrailtyEstimate
		'925831000000107', -- moderateFrailty,
		'1839051000006106', -- onFrailtyRegister,
		'910301000006104', -- frailPerson,
		'248279007', -- frailty,
		'908431000006104', -- frailElderlyPeople1,
		'909011000006109', -- frailElderlyPeople2,
		'2010291000006109', -- qFrailtyModerateFrailty,
		'2010331000006102', -- qFrailtyModerateFrailtyEstimate,
		'925791000000100', -- mildFrailty,
		'2010281000006106', -- qFrailtyMildFrailty,
		'2010321000006100' -- qFrailtyMildFrailtyEstimate
	) 
	ORDER BY result DESC
	LIMIT 1;

END$$
DELIMITER ;
