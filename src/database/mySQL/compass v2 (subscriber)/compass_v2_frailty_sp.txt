DELIMITER $$
CREATE PROCEDURE `get_frailty_flag`(
	_person_id bigint
)
BEGIN

	DROP TEMPORARY TABLE IF EXISTS tmp.frailty_patients;

	CREATE TEMPORARY TABLE tmp.frailty_patients AS
	SELECT id as `patient_id`
	FROM patient
	WHERE person_id = _person_id;

	CREATE INDEX ix ON tmp.frailty_patients (patient_id);

	DROP TEMPORARY TABLE IF EXISTS tmp.frailty_obs;

	CREATE TEMPORARY TABLE tmp.frailty_obs AS
	SELECT ob.*
	FROM tmp.frailty_patients p
	INNER JOIN observation ob
	ON ob.patient_id = p.patient_id;

	CREATE INDEX ix ON tmp.frailty_obs (non_core_concept_id);

	SELECT c.frailty_result
	FROM tmp.frailty_obs o
	INNER JOIN frailty_codes c
	ON c.concept_id = o.non_core_concept_id
	WHERE
		(c.min_value IS NULL
			OR (o.result_value IS NOT NULL AND o.result_value > c.min_value))
		AND (c.max_value IS NULL
			OR (o.result_value IS NOT NULL AND o.result_value <= c.max_value))
	ORDER BY frailty_result DESC -- frailty result has 1, 2, 3 so this will return the most severe
	LIMIT 1;


END$$
DELIMITER ;
